# Project Spire: the game, recompiled with the ReXGlue SDK.
#
# Build from the player's own default.xex: run `rexglue codegen fable_2_manifest.toml`
# first. SDK boilerplate lives in generated/rexglue.cmake.

cmake_minimum_required(VERSION 3.25)
project(fable_2 LANGUAGES CXX)

find_package(Python3 COMPONENTS Interpreter REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS constants/game_versions.json)
execute_process(COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_game_versions.py --check
    COMMAND_ERROR_IS_FATAL ANY)
include(constants/game_versions.cmake)
set(FABLE2_BUILD_PROFILE "${FABLE2_CATALOG_COMPATIBLE_PROFILE}" CACHE STRING "Verified guest XEX profile")
set_property(CACHE FABLE2_BUILD_PROFILE PROPERTY STRINGS
    "${FABLE2_CATALOG_COMPATIBLE_PROFILE}" ${FABLE2_CATALOG_PROFILES})
if(FABLE2_BUILD_PROFILE STREQUAL FABLE2_CATALOG_COMPATIBLE_PROFILE)
    set(FABLE2_SELECTED_PROFILES ${FABLE2_CATALOG_PROFILES})
elseif(FABLE2_BUILD_PROFILE IN_LIST FABLE2_CATALOG_PROFILES)
    set(FABLE2_SELECTED_PROFILES "${FABLE2_BUILD_PROFILE}")
else()
    message(FATAL_ERROR "Unverified FABLE2_BUILD_PROFILE: ${FABLE2_BUILD_PROFILE}")
endif()

set(CMAKE_CXX_STANDARD 23)
list(JOIN FABLE2_SELECTED_PROFILES "\", \"" FABLE2_PROFILES_JSON)
set(FABLE2_SUPPORTED_PROFILES "[\"${FABLE2_PROFILES_JSON}\"]")
configure_file(config/fable2_build.json.in ${CMAKE_CURRENT_BINARY_DIR}/fable2_build.json @ONLY)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(generated/rexglue.cmake)
set(FABLE2_CODEGEN_TOOL "" CACHE FILEPATH "Standalone compatible ReXGlue codegen tool")
if(FABLE2_CODEGEN_TOOL)
    if(NOT EXISTS "${FABLE2_CODEGEN_TOOL}")
        message(FATAL_ERROR "FABLE2_CODEGEN_TOOL does not exist")
    endif()
    # The official tool must keep its own runtime. Host headers, integration
    # sources and libraries come from the matched source-built SDK instead.
    set_target_properties(rex::rexglue PROPERTIES
        IMPORTED_LOCATION "${FABLE2_CODEGEN_TOOL}"
        IMPORTED_LOCATION_DEBUG "${FABLE2_CODEGEN_TOOL}"
        IMPORTED_LOCATION_RELWITHDEBINFO "${FABLE2_CODEGEN_TOOL}"
        IMPORTED_LOCATION_RELEASE "${FABLE2_CODEGEN_TOOL}")
endif()

# Sources
# Strong-symbol override copies of recompiled guest functions (perf-optimized;
# regenerated by tools/make_hotfunc_overrides.py - see file headers).
# Grouped into per-category subfolders (frame/, os_notifications/, refcount/, ...);
# glob recursively so new categories are picked up automatically.
file(GLOB_RECURSE HOTFUNC_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/core/hotfunc/*.cpp)

set(FABLE2_SOURCES
    src/main.cpp
    src/core/fable2_config.cpp
    src/core/fable2_patches.cpp
    src/core/fable2_hooks.cpp
    src/core/fable2_address_scan.cpp
    src/native_gpu/draw_capture.cpp
    src/native_gpu/options_menu.cpp
    src/input/mac_prompts.cpp
    src/input/mac_menu_mouse.cpp
    ${HOTFUNC_SOURCES}
)

add_executable(fable_2 ${FABLE2_SOURCES})

# The app's headers live in subfolders (src/core, src/diagnostics, src/input)
# and are included by bare filename ("fable2_config.h"), so the subfolders
# must be on the include path of both the app and the recompiled code.
set(FABLE2_HEADER_DIRS
    ${CMAKE_CURRENT_SOURCE_DIR}/src/core
    ${CMAKE_CURRENT_SOURCE_DIR}/src/diagnostics
    ${CMAKE_CURRENT_SOURCE_DIR}/src/input)
target_include_directories(fable_2 PRIVATE ${FABLE2_HEADER_DIRS})
if(APPLE)
    # Keycap prompts (src/input/mac_prompts.cpp).
    target_link_libraries(fable_2 PRIVATE "-framework CoreGraphics" "-framework CoreText"
                          "-framework CoreFoundation" z)
endif()
if(TARGET fable_2_recomp)
    target_include_directories(fable_2_recomp PRIVATE ${FABLE2_HEADER_DIRS})
endif()

# Stage the GPU plugin (Metal on macOS, Vulkan on Linux) next to the executable.
rexglue_setup_target(fable_2 GPU_PLUGINS xenos)
add_custom_command(TARGET fable_2 POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_CURRENT_BINARY_DIR}/fable2_build.json"
        "$<TARGET_FILE_DIR:fable_2>/fable2_build.json"
    COMMENT "Staging the supported game versions"
    VERBATIM)

target_compile_definitions(fable_2 PRIVATE
    FABLE2_BUILD_PROFILE="${FABLE2_BUILD_PROFILE}"
    FABLE2_NATIVE_CONFIGURATION="$<CONFIG>"
    FABLE2_NATIVE_SDK="$<IF:$<BOOL:${FABLE2_CODEGEN_TOOL}>,source-built,external>")

# Remote control server (AI input channel) is a debug-only tool: it opens a
# localhost TCP port and injects guest input, so it must NOT ship in Release /
# RelWithDebInfo builds (players shouldn't see a hidden input channel, and it's
# only useful for debugging). Defining FABLE2_REMOTE_CONTROL compiles the pad
# driver registration + server start/stop into the app; without it (non-Debug)
# those are #ifdef'd out entirely and no port is ever opened.
option(FABLE2_ENABLE_REMOTE_CONTROL "Compile the remote control server into non-Debug builds (local debugging only)" OFF)
if(CMAKE_BUILD_TYPE STREQUAL "Debug" OR FABLE2_ENABLE_REMOTE_CONTROL)
    target_compile_definitions(fable_2 PRIVATE FABLE2_REMOTE_CONTROL)
endif()

# --- Recomp-level patches (Xenia code-region patches, made effective) ------
# This is a RECOMP: guest .text is never executed, so Xenia code patches
# (byte writes to .text) are inert at runtime. tools/apply_recomp_patches.py
# mirrors those patches into codegen's emitted C++ instead, and runs between
# codegen and compiling fable_2_recomp (keyed on the codegen stamp), so the
# patches survive codegen re-runs. Missing anchors fail the build. Skip all
# with FABLE2_RECOMP_PATCHES=0 in the build environment (A/B runs).
if(TARGET fable_2_recomp)
    add_custom_command(
        OUTPUT "${CMAKE_CURRENT_SOURCE_DIR}/generated/default/recomp_patches.stamp"
        COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tools/apply_recomp_patches.py
                ${CMAKE_CURRENT_SOURCE_DIR}/generated/default
        COMMAND ${CMAKE_COMMAND} -E touch
                "${CMAKE_CURRENT_SOURCE_DIR}/generated/default/recomp_patches.stamp"
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/generated/default/codegen.build.stamp"
                ${CMAKE_CURRENT_SOURCE_DIR}/tools/apply_recomp_patches.py
        WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
        COMMENT "Applying recomp-level patches to generated code"
        VERBATIM
    )
    add_custom_target(fable_2_recomp_patches
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/generated/default/recomp_patches.stamp")
    add_dependencies(fable_2_recomp fable_2_recomp_patches)
endif()

# --- Guest function-call tracing (fable2_func_trace.log) -------------------
# Rexglue codegen emits REX_FUNC_PROLOGUE() at the top of every recompiled
# guest function. Force-include src/core/fable2_func_trace.h into fable_2_recomp:
# CMake orders a target's -include after its PCH include, so the header sees
# the generated pch already loaded and redefines REX_FUNC_PROLOGUE() to also
# call Fable2FuncTraceCall(__func__) - every guest function entry then logs
# its name (see the header). No generated files are modified, so codegen
# re-runs never lose the hook. Off by default; enable with
# FABLE2_FUNC_TRACE=1 (env) or Fable2FuncTraceSetEnabled(true) (runtime).
# Appended AFTER fable_2_pch.h in the target's precompile-header list: CMake
# writes the list into cmake_pch.hxx in order, so the hook section at the
# bottom of fable2_func_trace.h (which redefines REX_FUNC_PROLOGUE) runs with
# the generated pch already loaded, and the patched macro is baked into the
# PCH used by every recompiled TU.
if(TARGET fable_2_recomp)
    target_precompile_headers(fable_2_recomp PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/src/core/fable2_func_trace.h)
endif()

# The build never copies the game content (default.xex, data/). Point the game at
# it with --game_data_root=<path>; the launchers do this for an installed game.

# --- F5 external-Lua runner (src/lua/) -----------------------------------
# src/core/fable2_f5_lua.h runs an external script (cvar `f5_lua_path`, default
# "scripts/recomp/F5.lua") via CScriptManager::RunScript when F5 is pressed.
# The game's VFS indexes data/ from dir.manifest at startup, so the .lua file
# must exist AND be a dir.manifest line before the game starts. Sources live in
# src/lua/ (git-tracked); they are staged into BOTH the repo data/ (the
# git-tracked content root) and the exe dir (the default game data root), and
# dir.manifest is kept in sync in BOTH trees.
#
# This is a custom target that fable_2 depends on (NOT a POST_BUILD step) so it
# runs on every `build fable_2` even when the exe is up to date -- the .lua
# files are not CMake sources, so a no-op fable_2 relink would otherwise skip
# re-copying an edited script. dir.manifest syncing is idempotent: it appends
# only missing "scripts\\recomp\\<name>" entries (CRLF), leaving other entries
# untouched.
file(GLOB FABLE2_LUA_FILES ${CMAKE_CURRENT_SOURCE_DIR}/src/lua/*.lua)
add_custom_target(fable_2_lua_staging
  COMMAND ${CMAKE_COMMAND} -E copy_directory
          ${CMAKE_CURRENT_SOURCE_DIR}/src/lua
          ${CMAKE_CURRENT_SOURCE_DIR}/data/scripts/recomp
  COMMAND ${CMAKE_COMMAND} -E copy_directory
          ${CMAKE_CURRENT_SOURCE_DIR}/src/lua
          $<TARGET_FILE_DIR:fable_2>/data/scripts/recomp
  COMMAND ${CMAKE_COMMAND}
          -Dlua_dir=${CMAKE_CURRENT_SOURCE_DIR}/src/lua
          -Dmanifests=${CMAKE_CURRENT_SOURCE_DIR}/data/dir.manifest
          -P ${CMAKE_CURRENT_SOURCE_DIR}/tools/ensure_recomp_manifest.cmake
  COMMAND ${CMAKE_COMMAND}
          -Dlua_dir=${CMAKE_CURRENT_SOURCE_DIR}/src/lua
          -Dmanifests=$<TARGET_FILE_DIR:fable_2>/data/dir.manifest
          -P ${CMAKE_CURRENT_SOURCE_DIR}/tools/ensure_recomp_manifest.cmake
  DEPENDS ${FABLE2_LUA_FILES}
  COMMENT "Staging src/lua/* -> data/scripts/recomp/ + dir.manifest (content root + exe dir)"
)
add_dependencies(fable_2 fable_2_lua_staging)

# The remote-control CLI client is a debug-only feature (see the
# FABLE2_REMOTE_CONTROL gate above), so only stage it for Debug builds.
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  add_custom_command(TARGET fable_2 POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
            ${CMAKE_CURRENT_SOURCE_DIR}/tools/fable2_control.py
            $<TARGET_FILE_DIR:fable_2>/fable2_control.py
    COMMENT "Staging fable2_control.py (debug-only remote control client) next to fable_2"
  )
endif()

# (The screenshot command reads the game's rendered frame directly from the
# renderer's GPU guest output via Presenter::CaptureGuestOutput - see
# src/diagnostics/fable2_frame_capture.h - so no external capture helper is
# needed.)

# --- Recomp user config (fable2_config.toml) -------------------------------
# Header-only toml++ parser, vendored from the same version the ReXGlue SDK
# uses (thirdparty/rexglue-sdk-src/thirdparty/tomlplusplus), so the app's
# config loader does not depend on the SDK source tree being present.
target_include_directories(fable_2 SYSTEM PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/tomlplusplus/include)

# Stage the human-readable template next to the exe ONCE (configure time,
# copy-if-not-exists): it is user data, so user edits must survive rebuilds.
# The code also recreates it if it is ever deleted (keep that embedded copy
# in sync - see src/core/fable2_config.cpp).
if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/fable2_config.toml")
    file(COPY ${CMAKE_CURRENT_SOURCE_DIR}/config/fable2_config.toml
         DESTINATION ${CMAKE_CURRENT_BINARY_DIR})
endif()

# Keep the checkout path out of the binary (__FILE__ in assert messages).
target_compile_options(fable_2 PRIVATE -ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=)
if(TARGET fable_2_recomp)
    target_compile_options(fable_2_recomp PRIVATE -ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=)
endif()
